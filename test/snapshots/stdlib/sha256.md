{- | SHA-256 hashing of a byte array (FIPS 180-4).

   A message is an `Array Int` with each element from `0` to `255`, the same
   form `readFileBytes` and `hex.decode` use. The digest is 32 bytes, most
   significant byte of each word first.

   `sha256` panics when any element is outside `0` to `255`. Reducing such an
   element modulo 256 would map two different messages onto one digest, which
   a content-addressing caller has no way to detect. `sha256FixedBytes` is
   the same hash without that scan, for a caller whose bytes are already
   known to be in range.

   There is no incremental form. A message is hashed in one call. -}

-- Arithmetic contract. Every word is an `Int` in [0, 2^32). `mask32 x =
-- bitAnd x 4294967295`, and every `+`, `shiftLeft` and `bitNot` result is
-- immediately followed by `mask32`. `shiftRight` needs no mask: it is
-- logical/unsigned (`stdlib/runtime.mdk`) and its input is already masked. A
-- sum of up to five 32-bit values is < 2^35, well inside the 63-bit fixnum
-- (`stdlib/bits64.mdk`), so no intermediate overflows; the mask enforces the
-- modulus, not fixnum safety.
--
-- Message words are big-endian (FIPS 180-4 §3.1) and the digest is emitted in
-- H0..H7 order.
--
-- No padded copy of the whole message is ever materialized: full 64-byte
-- blocks are read directly from the input by offset, and only the final one
-- or two blocks are synthesized to hold the 0x80 marker, the zero padding and
-- the 64-bit big-endian bit length.
--
-- Writes go through `array.setInPlace` rather than the `arr[i] := v` sugar.
-- The two now emit and inline identically, so migrating is mechanical;
-- re-measure the hot loops when you do.
import array.{get, setInPlace, make, blit}

-- ── Bit / word arithmetic ────────────────────────────────────────────

mask32 : Int -> Int
mask32 x = bitAnd x 4294967295

rotr32 : Int -> Int -> Int
rotr32 n x = mask32 (bitOr (shiftRight x n) (shiftLeft x (32 - n)))

ch : Int -> Int -> Int -> Int
ch x y z = bitXor (bitAnd x y) (bitAnd (mask32 (bitNot x)) z)

maj : Int -> Int -> Int -> Int
maj x y z = bitXor (bitXor (bitAnd x y) (bitAnd x z)) (bitAnd y z)

bigSigma0 : Int -> Int
bigSigma0 x = bitXor (bitXor (rotr32 2 x) (rotr32 13 x)) (rotr32 22 x)

bigSigma1 : Int -> Int
bigSigma1 x = bitXor (bitXor (rotr32 6 x) (rotr32 11 x)) (rotr32 25 x)

smallSigma0 : Int -> Int
smallSigma0 x = bitXor (bitXor (rotr32 7 x) (rotr32 18 x)) (shiftRight x 3)

smallSigma1 : Int -> Int
smallSigma1 x = bitXor (bitXor (rotr32 17 x) (rotr32 19 x)) (shiftRight x 10)

-- In-bounds indexing via the safe `Array.get`, panicking on a miss. Every
-- call site below only ever indexes within a known length, so the `None`
-- arm is unreachable (same idiom as `stdlib/hex.mdk`'s `byteAt`).
byteAt : Array Int -> Int -> Int
byteAt arr i = match get i arr
  Some b => b
  None => panic "sha256: byteAt index out of bounds"

wordAt : Array Int -> Int -> Int
wordAt arr i = match get i arr
  Some w => w
  None => panic "sha256: wordAt index out of bounds"

-- ── Message schedule ─────────────────────────────────────────────────

-- Big-endian 32-bit word starting at byte index `i*4`, read through an
-- arbitrary byte accessor (so the caller decides whether bytes come from
-- the original message or a synthesized tail block).
scheduleWord : (Int -> Int) -> Int -> Int
scheduleWord byteOf i =
  let b0 = byteOf (i * 4)
  let b1 = byteOf (i * 4 + 1)
  let b2 = byteOf (i * 4 + 2)
  let b3 = byteOf (i * 4 + 3)
  mask32
    (bitOr
      (bitOr (bitOr (shiftLeft b0 24) (shiftLeft b1 16)) (shiftLeft b2 8))
      b3)

fillFirst16 : Array Int -> (Int -> Int) -> Int -> Unit
fillFirst16 w byteOf i =
  if i >= 16 then
    ()
  else
    setInPlace i (scheduleWord byteOf i) w
    fillFirst16 w byteOf (i + 1)

extendSchedule : Array Int -> Int -> Unit
extendSchedule w i =
  if i >= 64 then
    ()
  else
    let s0 = smallSigma0 (wordAt w (i - 15))
    let s1 = smallSigma1 (wordAt w (i - 2))
    let v =
      mask32 (mask32 (mask32 (wordAt w (i - 16) + s0) + wordAt w (i - 7)) + s1)
    setInPlace i v w
    extendSchedule w (i + 1)

-- Build the 64-word message schedule for one block from a byte accessor
-- covering that block's 64 bytes (offsets 0..63).
buildSchedule : (Int -> Int) -> Array Int
buildSchedule byteOf =
  let w = make 64 0
  fillFirst16 w byteOf 0
  extendSchedule w 16
  w

-- ── Compression ──────────────────────────────────────────────────────

type State = (Int, Int, Int, Int, Int, Int, Int, Int)

compressRounds : Array Int -> State -> Int -> State
compressRounds w (a, b, c, d, e, f, g, h) t =
  if t >= 64 then
    (a, b, c, d, e, f, g, h)
  else
    let t1 =
      mask32
        (mask32 (mask32 (mask32 (h + bigSigma1 e) + ch e f g) + wordAt k t)
          + wordAt w t)
    let t2 = mask32 (bigSigma0 a + maj a b c)
    let a' = mask32 (t1 + t2)
    let e' = mask32 (d + t1)
    compressRounds w (a', a, b, c, e', e, f, g) (t + 1)

-- One 64-byte block, addressed via `byteOf` (offsets 0..63 relative to the
-- block start), folded into the running hash state.
compressBlock : (Int -> Int) -> State -> State
compressBlock byteOf (h0, h1, h2, h3, h4, h5, h6, h7) =
  let w = buildSchedule byteOf
  let (a, b, c, d, e, f, g, h) =
    compressRounds w (h0, h1, h2, h3, h4, h5, h6, h7) 0
  (
    mask32 (h0 + a),
    mask32 (h1 + b),
    mask32 (h2 + c),
    mask32 (h3 + d),
    mask32 (h4 + e),
    mask32 (h5 + f),
    mask32 (h6 + g),
    mask32 (h7 + h),
  )

-- ── Driving the blocks ───────────────────────────────────────────────

-- Fold every FULL 64-byte block of `msg` directly by offset — no copy of
-- the message is ever made.
processFullBlocks : Array Int -> Int -> State -> Int -> State
processFullBlocks msg numFull st blockIdx =
  if blockIdx >= numFull then
    st
  else
    let base = blockIdx * 64
    processFullBlocks
      msg
      numFull
      (compressBlock (j => byteAt msg (base + j)) st)
      (blockIdx + 1)

-- The length, in bits, of `padLength`-byte messages fits big-endian into
-- the last 8 bytes of the padded block: writes it there.
writeLenGo : Array Int -> Int -> Int -> Int -> Unit
writeLenGo out offset bitLen i =
  if i >= 8 then
    ()
  else
    let shiftAmt = (7 - i) * 8
    setInPlace (offset + i) (bitAnd (shiftRight bitLen shiftAmt) 255) out
    writeLenGo out offset bitLen (i + 1)

-- The number of bytes in the final, padded region (one block if the tail
-- plus the 0x80 marker plus the 8-byte length fits in 64 bytes, else two).
padLength : Int -> Int
padLength tailLen = if tailLen + 1 + 8 <= 64 then 64 else 128

-- Synthesize the final one-or-two padded blocks: `tailLen` message bytes
-- (copied via `blit`, never the whole message), then `0x80`, then zero
-- bytes, then the 64-bit big-endian bit length of the WHOLE message.
buildTail : Array Int -> Int -> Int -> Array Int
buildTail msg len tailLen =
  let size = padLength tailLen
  let out = make size 0
  blit msg (len - tailLen) out 0 tailLen
  setInPlace tailLen 0x80 out
  writeLenGo out (size - 8) (len * 8) 0
  out

processTail : Array Int -> State -> State
processTail tail st =
  let numBlocks = arrayLength tail / 64
  processFullBlocks tail numBlocks st 0

-- ── Public entry point ───────────────────────────────────────────────

-- Is every element of `msg[0 .. len)` inside the `0..255` byte domain? A
-- tail-recursive index scan over safe indexing, allocating nothing.
--
-- This must stay a full-arity self tail call. Native lowers that shape to
-- `musttail`, and callers hash messages of a megabyte and more, so a
-- non-tail rewrite here is a million-frame stack overflow rather than a
-- slowdown.
byteDomainOk : Array Int -> Int -> Int -> Bool
byteDomainOk msg len i =
  if i >= len then
    True
  else
    let b = msg[i]
    if b < 0 || b > 255 then False else byteDomainOk msg len (i + 1)

sha256AssumeByteDomain : Array Int -> Array Int
sha256AssumeByteDomain msg =
  let len = arrayLength msg
  let numFull = len / 64
  let tailLen = len - numFull * 64
  let afterFull = processFullBlocks msg numFull h0Init 0
  let tail = buildTail msg len tailLen
  let (h0, h1, h2, h3, h4, h5, h6, h7) = processTail tail afterFull
  digestBytes (h0, h1, h2, h3, h4, h5, h6, h7)

-- This entry exists for `hmac`, whose every input byte comes from a fixed
-- pad, a SHA-256 digest or caller-canonical scalar bytes. Running the
-- byte-domain scan over those would put a secret-derived early exit inside
-- the signing closure.
{- | The 32-byte SHA-256 digest of `msg`, skipping the byte-domain check.

   Every element of `msg` must still be in `0` to `255`. Nothing here checks
   that, and an element outside the range silently digests as some other
   message. Prefer `sha256` unless the bytes come from a source that already
   guarantees the range. -}
export
sha256FixedBytes : Array Int -> Array Int
sha256FixedBytes msg = sha256AssumeByteDomain msg

{- | The 32-byte SHA-256 digest of `msg`.

   Panics when any element of `msg` is outside `0` to `255`. -}
export
sha256 : Array Int -> Array Int
sha256 msg =
  let len = arrayLength msg
  if not (byteDomainOk msg len 0) then
    panic "sha256: every element must be a byte in 0..255"
  else
    sha256AssumeByteDomain msg

digestWord : Array Int -> Int -> Int -> Unit
digestWord out i word =
  setInPlace (i * 4) (bitAnd (shiftRight word 24) 255) out
  setInPlace (i * 4 + 1) (bitAnd (shiftRight word 16) 255) out
  setInPlace (i * 4 + 2) (bitAnd (shiftRight word 8) 255) out
  setInPlace (i * 4 + 3) (bitAnd word 255) out

digestBytes : State -> Array Int
digestBytes (h0, h1, h2, h3, h4, h5, h6, h7) =
  let out = make 32 0
  digestWord out 0 h0
  digestWord out 1 h1
  digestWord out 2 h2
  digestWord out 3 h3
  digestWord out 4 h4
  digestWord out 5 h5
  digestWord out 6 h6
  digestWord out 7 h7
  out
