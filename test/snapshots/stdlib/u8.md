{- | Unsigned 8-bit integers, the `U8` type.

   A `U8` holds a value from `0` to `255`. Its arithmetic wraps: `+`, `-`
   and `*` give the result modulo 2^8, so `255 + 1` is `0` and `0 - 1` is
   `255`. `/` and `%` are unsigned division and remainder, and a zero
   divisor panics as it does for `Int`.

   An integer literal in `U8` position must fit, so `(256 : U8)` is a
   compile-time error. Converting a computed `Int` never wraps silently:
   `fromInt` panics when the value is out of range, `tryFromInt` answers
   `None`, and `truncate` keeps the low 8 bits and is the only conversion
   that discards any. `toInt` goes the other way and always succeeds.

   Within the family, `truncateU16` and `truncateU32` keep the low 8 bits
   of a wider one.

   `bitAnd`, `bitOr`, `bitXor`, `bitNot`, `shiftLeft` and `shiftRight` have
   the same names as the prelude's `Int` operations. Import the module
   qualified, as `import u8 as U8`, and write `U8.shiftLeft`.

   `U8` values compare, hash, and render in decimal like `Int`. `toHex`
   renders the 2-digit hexadecimal form. -}

-- Every operation keeps a value in 0 .. 2^n - 1: the kernel conversions
-- (`u8Truncate`, `u8ToInt`) and the `Int` bit primitives under their
-- `int*` names are internal externs (stdlib/runtime.mdk), reached here because
-- this module's own `bitAnd`, `shiftLeft` and the rest shadow the prelude's.

-- # Conversions

{- | The value as a `U8`, or `None` when it is outside `0` to `255`.

   > tryFromInt 200 |> map toInt
   Some 200
   > tryFromInt 256 |> map toInt
   None -}
export
tryFromInt : Int -> Option U8
tryFromInt n = if n < 0 || n > 255 then None else Some (u8Truncate n)

-- > tryFromInt (-1) |> map toInt
-- None

{- | The low 8 bits of the value, as a `U8`.

   This is the one conversion that discards bits: a value outside `0` to
   `255` is reduced modulo 2^8, and a negative value is taken in two's
   complement.

   > toInt (truncate 257)
   1
   > toInt (truncate (-1))
   255 -}
export
truncate : Int -> U8
truncate n = u8Truncate n

-- | The value as an `Int`, from `0` to `255`.
export
toInt : U8 -> Int
toInt x = u8ToInt x

-- > toInt (255 : U8)
-- 255

-- | The low 8 bits of a `U16`, as a `U8`.
export
truncateU16 : U16 -> U8
truncateU16 x = u8Truncate (u16ToInt x)

-- | The low 8 bits of a `U32`, as a `U8`.
export
truncateU32 : U32 -> U8
truncateU32 x = u8Truncate (u32ToInt x)

-- # Bit operations

{- | The bitwise and of two values.

   > toInt (bitAnd 12 10)
   8 -}
export
bitAnd : U8 -> U8 -> U8
bitAnd a b = u8Truncate (intBitAnd (u8ToInt a) (u8ToInt b))

{- | The bitwise or of two values.

   > toInt (bitOr 12 10)
   14 -}
export
bitOr : U8 -> U8 -> U8
bitOr a b = u8Truncate (intBitOr (u8ToInt a) (u8ToInt b))

{- | The bitwise exclusive or of two values.

   > toInt (bitXor 12 10)
   6 -}
export
bitXor : U8 -> U8 -> U8
bitXor a b = u8Truncate (intBitXor (u8ToInt a) (u8ToInt b))

{- | Every bit of the value inverted, within 8 bits.

   > toInt (bitNot 0)
   255 -}
export
bitNot : U8 -> U8
bitNot x = u8Truncate (intBitNot (u8ToInt x))

{- | The value shifted left by a number of bits, the vacated bits zero.

   Bits shifted past bit 7 are discarded, and a shift of 8 or more
   gives `0`. A negative shift panics.

   > toInt (shiftLeft 1 4)
   16
   > toInt (shiftLeft 1 8)
   0 -}
export
shiftLeft : U8 -> Int -> U8
shiftLeft x k
  | k < 0 = panic "u8.shiftLeft: negative shift \{k}"
  | k >= 8 = u8Truncate 0
  | otherwise = u8Truncate (intShiftLeft (u8ToInt x) k)

-- > toInt (shiftLeft 255 1)
-- 254

{- | The value shifted right by a number of bits, the vacated bits zero.

   A shift of 8 or more gives `0`. A negative shift panics.

   > toInt (shiftRight 16 4)
   1 -}
export
shiftRight : U8 -> Int -> U8
shiftRight x k
  | k < 0 = panic "u8.shiftRight: negative shift \{k}"
  | k >= 8 = u8Truncate 0
  | otherwise = u8Truncate (intShiftRight (u8ToInt x) k)

-- the rotate amount reduced into 0 .. 7, negative amounts included.
rotateAmount : Int -> Int
rotateAmount k = (k % 8 + 8) % 8

{- | The value rotated left by a number of bits: the bits shifted out at the
   top come back in at the bottom.

   The amount is taken modulo 8, so a negative amount rotates right.

   > toInt (rotateLeft 128 1)
   1 -}
export
rotateLeft : U8 -> Int -> U8
rotateLeft x k =
  let r = rotateAmount k
  if r == 0 then x else bitOr (shiftLeft x r) (shiftRight x (8 - r))

-- > toInt (rotateLeft 1 (-1))
-- 128

{- | The value rotated right by a number of bits: the bits shifted out at
   the bottom come back in at the top.

   The amount is taken modulo 8, so a negative amount rotates left.

   > toInt (rotateRight 1 1)
   128 -}
export
rotateRight : U8 -> Int -> U8
rotateRight x k = rotateLeft x (8 - rotateAmount k)

-- > toInt (rotateRight 1 8)
-- 1

-- the number of one bits in an Int from 0 upward, counted a bit at a time.
popCountFrom : Int -> Int -> Int
popCountFrom acc n = if n == 0 then acc else popCountFrom (acc + n % 2) (n / 2)

{- | The number of one bits.

   > popCount 255
   8 -}
export
popCount : U8 -> Int
popCount x = popCountFrom 0 (u8ToInt x)

-- the number of bits needed to write an Int from 0 upward.
bitLengthFrom : Int -> Int -> Int
bitLengthFrom acc n = if n == 0 then acc else bitLengthFrom (acc + 1) (n / 2)

{- | The number of zero bits above the highest one bit, 8 for `0`.

   > leadingZeros 1
   7 -}
export
leadingZeros : U8 -> Int
leadingZeros x = 8 - bitLengthFrom 0 (u8ToInt x)

-- the number of zero bits below the lowest one bit of a positive Int.
trailingZerosFrom : Int -> Int -> Int
trailingZerosFrom acc n =
  if n % 2 == 1 then acc else trailingZerosFrom (acc + 1) (n / 2)

{- | The number of zero bits below the lowest one bit, 8 for `0`.

   > trailingZeros 8
   3 -}
export
trailingZeros : U8 -> Int
trailingZeros x =
  let n = u8ToInt x
  if n == 0 then 8 else trailingZerosFrom 0 n

-- > trailingZeros 0
-- 8
-- > leadingZeros 0
-- 8

-- # Rendering

-- the lowercase hex digit for a value from 0 to 15.
hexDigit : Int -> String
hexDigit d =
  charToStr (optionOr '0' (charFromCode (if d < 10 then 48 + d else 87 + d)))

{- | The value in hexadecimal: 2 lowercase digits, zero-padded, with no
   prefix.

   > toHex 255
   "ff" -}
export
toHex : U8 -> String
toHex x = hexDigit (u8ToInt x / 16) ++ hexDigit (u8ToInt x % 16)

-- # Instances

export impl Eq U8 where
  eq a b = a == b

-- Every method is defined, not only `compare`, so each one is a single builtin
-- comparison and a generic caller never falls back to an interface default.
-- `min` and `max` are the methods being defined, so they are written out.
-- lint-disable-next-line rule-if-max-min
export impl Ord U8 where
  compare a b = if a < b then Lt else if a > b then Gt else Eq
  lt a b = a < b
  gt a b = a > b
  lte a b = a <= b
  gte a b = a >= b
  min a b = if a <= b then a else b
  max a b = if a >= b then a else b

{- | Arithmetic modulo 2^8. `fromInt` panics when the `Int` is outside `0`
   to `255`; it is what a literal in a generic `Num a =>` position becomes,
   and `tryFromInt` and `truncate` are the non-panicking conversions. -}
export impl Num U8 where
  add a b = a + b
  sub a b = a - b
  mul a b = a * b
  div a b = a / b
  negate a = 0 - a
  abs a = a
  signum a = if a == 0 then 0 else 1
  fromInt n =
    if n < 0 || n > 255 then
      panic "fromInt: \{n} does not fit U8 (0..255)"
    else
      u8Truncate n
  rem a b = a % b

export impl Bounded U8 where
  minBound = u8Truncate 0
  maxBound = u8Truncate 255

export impl Hashable U8 where
  hash x = hashInt (u8ToInt x)

export impl Display U8 where
  display x = intToString (u8ToInt x)

export impl Debug U8 where
  debug x = intToString (u8ToInt x)
