{- | Unsigned 16-bit integers, the `U16` type.

   A `U16` holds a value from `0` to `65535`. Its arithmetic wraps: `+`,
   `-` and `*` give the result modulo 2^16, so `65535 + 1` is `0` and `0 -
   1` is `65535`. `/` and `%` are unsigned division and remainder, and a
   zero divisor panics as it does for `Int`.

   An integer literal in `U16` position must fit, so `(65536 : U16)` is a
   compile-time error. Converting a computed `Int` never wraps silently:
   `fromInt` panics when the value is out of range, `tryFromInt` answers
   `None`, and `truncate` keeps the low 16 bits and is the only conversion
   that discards any. `toInt` goes the other way and always succeeds.

   Within the family, `fromU8` widens a narrower value, and `truncateU32`
   keeps the low 16 bits of a wider one.

   `bitAnd`, `bitOr`, `bitXor`, `bitNot`, `shiftLeft` and `shiftRight` have
   the same names as the prelude's `Int` operations. Import the module
   qualified, as `import u16 as U16`, and write `U16.shiftLeft`.

   `toBytesBE` and `toBytesLE` write a value as 2 bytes, most or least
   significant first, and `fromBytesBE` and `fromBytesLE` read one back.

   `U16` values compare, hash, and render in decimal like `Int`. `toHex`
   renders the 4-digit hexadecimal form. -}

-- Every operation keeps a value in 0 .. 2^n - 1: the kernel conversions
-- (`u16Truncate`, `u16ToInt`) and the `Int` bit primitives under their
-- `int*` names are internal externs (stdlib/runtime.mdk), reached here because
-- this module's own `bitAnd`, `shiftLeft` and the rest shadow the prelude's.

import u8 as U8
import bytes.{Bytes}
import bytes as B

-- # Conversions

{- | The value as a `U16`, or `None` when it is outside `0` to `65535`.

   > tryFromInt 200 |> map toInt
   Some 200
   > tryFromInt 65536 |> map toInt
   None -}
export
tryFromInt : Int -> Option U16
tryFromInt n = if n < 0 || n > 65535 then None else Some (u16Truncate n)

-- > tryFromInt (-1) |> map toInt
-- None

{- | The low 16 bits of the value, as a `U16`.

   This is the one conversion that discards bits: a value outside `0` to
   `65535` is reduced modulo 2^16, and a negative value is taken in two's
   complement.

   > toInt (truncate 65537)
   1
   > toInt (truncate (-1))
   65535 -}
export
truncate : Int -> U16
truncate n = u16Truncate n

-- | The value as an `Int`, from `0` to `65535`.
export
toInt : U16 -> Int
toInt x = u16ToInt x

-- > toInt (65535 : U16)
-- 65535

{- | A `U8` as a `U16`, the same number.

   > toInt (fromU8 (255 : U8))
   255 -}
export
fromU8 : U8 -> U16
fromU8 x = u16Truncate (u8ToInt x)

-- | The low 16 bits of a `U32`, as a `U16`.
export
truncateU32 : U32 -> U16
truncateU32 x = u16Truncate (u32ToInt x)

-- # Bit operations

{- | The bitwise and of two values.

   > toInt (bitAnd 12 10)
   8 -}
export
bitAnd : U16 -> U16 -> U16
bitAnd a b = u16Truncate (intBitAnd (u16ToInt a) (u16ToInt b))

{- | The bitwise or of two values.

   > toInt (bitOr 12 10)
   14 -}
export
bitOr : U16 -> U16 -> U16
bitOr a b = u16Truncate (intBitOr (u16ToInt a) (u16ToInt b))

{- | The bitwise exclusive or of two values.

   > toInt (bitXor 12 10)
   6 -}
export
bitXor : U16 -> U16 -> U16
bitXor a b = u16Truncate (intBitXor (u16ToInt a) (u16ToInt b))

{- | Every bit of the value inverted, within 16 bits.

   > toInt (bitNot 0)
   65535 -}
export
bitNot : U16 -> U16
bitNot x = u16Truncate (intBitNot (u16ToInt x))

{- | The value shifted left by a number of bits, the vacated bits zero.

   Bits shifted past bit 15 are discarded, and a shift of 16 or more
   gives `0`. A negative shift panics.

   > toInt (shiftLeft 1 4)
   16
   > toInt (shiftLeft 1 16)
   0 -}
export
shiftLeft : U16 -> Int -> U16
shiftLeft x k
  | k < 0 = panic "u16.shiftLeft: negative shift \{k}"
  | k >= 16 = u16Truncate 0
  | otherwise = u16Truncate (intShiftLeft (u16ToInt x) k)

-- > toInt (shiftLeft 65535 1)
-- 65534

{- | The value shifted right by a number of bits, the vacated bits zero.

   A shift of 16 or more gives `0`. A negative shift panics.

   > toInt (shiftRight 16 4)
   1 -}
export
shiftRight : U16 -> Int -> U16
shiftRight x k
  | k < 0 = panic "u16.shiftRight: negative shift \{k}"
  | k >= 16 = u16Truncate 0
  | otherwise = u16Truncate (intShiftRight (u16ToInt x) k)

-- the rotate amount reduced into 0 .. 15, negative amounts included.
rotateAmount : Int -> Int
rotateAmount k = (k % 16 + 16) % 16

{- | The value rotated left by a number of bits: the bits shifted out at the
   top come back in at the bottom.

   The amount is taken modulo 16, so a negative amount rotates right.

   > toInt (rotateLeft 32768 1)
   1 -}
export
rotateLeft : U16 -> Int -> U16
rotateLeft x k =
  let r = rotateAmount k
  if r == 0 then x else bitOr (shiftLeft x r) (shiftRight x (16 - r))

-- > toInt (rotateLeft 1 (-1))
-- 32768

{- | The value rotated right by a number of bits: the bits shifted out at
   the bottom come back in at the top.

   The amount is taken modulo 16, so a negative amount rotates left.

   > toInt (rotateRight 1 1)
   32768 -}
export
rotateRight : U16 -> Int -> U16
rotateRight x k = rotateLeft x (16 - rotateAmount k)

-- > toInt (rotateRight 1 16)
-- 1

-- the high and low 8 bits of the value, each as a `U8`.
highHalf : U16 -> U8
highHalf x = u8Truncate (intShiftRight (u16ToInt x) 8)

lowHalf : U16 -> U8
lowHalf x = u8Truncate (u16ToInt x)

{- | The number of one bits.

   > popCount 255
   8 -}
export
popCount : U16 -> Int
popCount x = U8.popCount (highHalf x) + U8.popCount (lowHalf x)

{- | The number of zero bits above the highest one bit, 16 for `0`.

   > leadingZeros 1
   15 -}
export
leadingZeros : U16 -> Int
leadingZeros x =
  let hi = highHalf x
  if u8ToInt hi == 0 then
    8 + U8.leadingZeros (lowHalf x)
  else
    U8.leadingZeros hi

{- | The number of zero bits below the lowest one bit, 16 for `0`.

   > trailingZeros 8
   3 -}
export
trailingZeros : U16 -> Int
trailingZeros x =
  let lo = lowHalf x
  if u8ToInt lo == 0 then
    8 + U8.trailingZeros (highHalf x)
  else
    U8.trailingZeros lo

-- > trailingZeros 0
-- 16
-- > leadingZeros 0
-- 16

-- # Bytes

-- every element is a byte by construction, so `fromArray` cannot refuse it.
bytesOf : Array Int -> Bytes
bytesOf bs = match B.fromArray bs
  Some b => b
  None => panic "u16.toBytes: a byte out of range"

-- the value whose bytes are [hi] and [lo], when the string is exactly two long.
fromTwo : Bytes -> Int -> Int -> Option U16
fromTwo b hi lo = match (B.length b, B.get hi b, B.get lo b)
  (2, Some h, Some l) => Some (u16Truncate (h * 256 + l))
  _ => None

{- | The value as 2 bytes, most significant first.

   > B.toArray (toBytesBE 258)
   [|1, 2|] -}
export
toBytesBE : U16 -> Bytes
toBytesBE x =
  let n = u16ToInt x
  bytesOf [|intShiftRight n 8, intBitAnd n 255|]

{- | The value as 2 bytes, least significant first.

   > B.toArray (toBytesLE 258)
   [|2, 1|] -}
export
toBytesLE : U16 -> Bytes
toBytesLE x =
  let n = u16ToInt x
  bytesOf [|intBitAnd n 255, intShiftRight n 8|]

{- | The value written as exactly 2 bytes, most significant first, or
   `None` when the byte string has any other length.

   > fromBytesBE (toBytesBE 258) |> map toInt
   Some 258 -}
export
fromBytesBE : Bytes -> Option U16
fromBytesBE b = fromTwo b 0 1

{- | The value written as exactly 2 bytes, least significant first, or
   `None` when the byte string has any other length.

   > fromBytesLE (toBytesLE 258) |> map toInt
   Some 258 -}
export
fromBytesLE : Bytes -> Option U16
fromBytesLE b = fromTwo b 1 0

-- > fromBytesBE (toBytesBE 1 ++ toBytesBE 1)
-- None

-- # Rendering

{- | The value in hexadecimal: 4 lowercase digits, zero-padded, with no
   prefix.

   > toHex 255
   "00ff" -}
export
toHex : U16 -> String
toHex x = U8.toHex (highHalf x) ++ U8.toHex (lowHalf x)

-- # Instances

export impl Eq U16 where
  eq a b = a == b

-- Every method is defined, not only `compare`: each is then the builtin word
-- compare, and no generic caller reaches an interface default.  `min` and
-- `max` are defined here, so they cannot be written with themselves.
-- lint-disable-next-line rule-if-max-min
export impl Ord U16 where
  compare a b = if a < b then Lt else if a > b then Gt else Eq
  lt a b = a < b
  gt a b = a > b
  lte a b = a <= b
  gte a b = a >= b
  min a b = if a <= b then a else b
  max a b = if a >= b then a else b

{- | Arithmetic modulo 2^16. `fromInt` panics when the `Int` is outside `0`
   to `65535`; it is what a literal in a generic `Num a =>` position becomes,
   and `tryFromInt` and `truncate` are the non-panicking conversions. -}
export impl Num U16 where
  add a b = a + b
  sub a b = a - b
  mul a b = a * b
  div a b = a / b
  negate a = 0 - a
  abs a = a
  signum a = if a == 0 then 0 else 1
  fromInt n =
    if n < 0 || n > 65535 then
      panic "fromInt: \{n} does not fit U16 (0..65535)"
    else
      u16Truncate n
  rem a b = a % b

export impl Bounded U16 where
  minBound = u16Truncate 0
  maxBound = u16Truncate 65535

export impl Hashable U16 where
  hash x = hashInt (u16ToInt x)

export impl Display U16 where
  display x = intToString (u16ToInt x)

export impl Debug U16 where
  debug x = intToString (u16ToInt x)
