{- | Signed 64-bit integers, the `I64` type.

   An `I64` holds a value from `-9223372036854775808` to
   `9223372036854775807`. Its arithmetic wraps in two's complement: `+`, `-`
   and `*` give the result modulo 2^64, so `9223372036854775807 + 1` is
   `-9223372036854775808`. `/` and `%` truncate toward zero, as C does, so
   `-7 / 2` is `-3` and `-7 % 2` is `-1`; `minBound / -1` wraps to `minBound`,
   and a zero divisor panics as it does for `Int`. `negate` and `abs` wrap the
   same way, so both give `minBound` for `minBound`.

   An integer literal in `I64` position may be as large as the type holds,
   including beyond `Int`'s range, as in `-9223372036854775808`. Every `Int`
   fits, so `fromInt` never panics and needs no checked or truncating
   counterpart.

   `Int` holds 63 bits, so the way back narrows: `toInt` answers `None` for a
   value outside `-4611686018427387904` to `4611686018427387903`.

   `fromU8`, `fromU16`, `fromU32` and `fromI32` widen a narrower value.
   `fromBits` and `toBits` convert to and from `U64` by keeping the 64 bits and
   changing how they are read, so the bit operations the `u64` module has and
   this one does not (rotations, bit counts, bytes, hexadecimal) are reached
   through them.

   `bitAnd`, `bitOr`, `bitXor`, `bitNot`, `shiftLeft` and `shiftRight` have the
   same names as the prelude's `Int` operations. Import the module qualified,
   as `import i64 as I64`, and write `I64.shiftLeft`. `shiftRight` is
   arithmetic: the vacated bits copy the sign.

   `I64` values compare, hash, and render in decimal like `Int`. -}

-- An `I64` is a boxed 64-bit cell, not a tagged word, with a header of its own.
-- Its operators are builtin; the kernels `i64FromBits` and `i64ToBits` are
-- internal externs (stdlib/runtime.mdk) that retype the 64 bits, and everything
-- else here is built on those and the `U64` kernels.

import u8
import u16
import u32
import u64
import i32

-- An `Int` as an `I64`: `u64Truncate` takes it in two's complement, which is
-- the same 64 bits.
ofInt : Int -> I64
ofInt n = i64FromBits (u64Truncate n)

-- # Conversions

{- | The value as an `Int`, or `None` when it is outside `-4611686018427387904`
   to `4611686018427387903`.

   > toInt (-5)
   Some -5
   > toInt 9223372036854775807
   None -}
export
toInt : I64 -> Option Int
toInt x =
  if x < -4611686018427387904 || x > 4611686018427387903 then
    None
  else
    Some (u64TruncateToInt (i64ToBits x))

-- > toInt (-4611686018427387904)
-- Some -4611686018427387904
-- > toInt (-4611686018427387905)
-- None

{- | A `U8` as an `I64`, the same number.

   > toInt (fromU8 (255 : U8))
   Some 255 -}
export
fromU8 : U8 -> I64
fromU8 x = ofInt (u8ToInt x)

{- | A `U16` as an `I64`, the same number.

   > toInt (fromU16 (65535 : U16))
   Some 65535 -}
export
fromU16 : U16 -> I64
fromU16 x = ofInt (u16ToInt x)

{- | A `U32` as an `I64`, the same number.

   > toInt (fromU32 (4294967295 : U32))
   Some 4294967295 -}
export
fromU32 : U32 -> I64
fromU32 x = ofInt (u32ToInt x)

{- | An `I32` as an `I64`, the same number.

   > toInt (fromI32 (-2147483648 : I32))
   Some -2147483648 -}
export
fromI32 : I32 -> I64
fromI32 x = ofInt (i32ToInt x)

{- | A `U64`'s 64 bits read in two's complement: the same bits, a different
   number when the top bit is set.

   > toInt (fromBits (0xFFFFFFFFFFFFFFFF : U64))
   Some -1 -}
export
fromBits : U64 -> I64
fromBits x = i64FromBits x

{- | The value's 64 bits in two's complement, as a `U64`.

   > toBits (-1) == 0xFFFFFFFFFFFFFFFF
   True -}
export
toBits : I64 -> U64
toBits x = i64ToBits x

-- # Bit operations

{- | The bitwise and of two values.

   > toInt (bitAnd (-4) 7)
   Some 4 -}
export
bitAnd : I64 -> I64 -> I64
bitAnd a b = i64FromBits (u64BitAnd (i64ToBits a) (i64ToBits b))

{- | The bitwise or of two values.

   > toInt (bitOr 12 10)
   Some 14 -}
export
bitOr : I64 -> I64 -> I64
bitOr a b = i64FromBits (u64BitOr (i64ToBits a) (i64ToBits b))

{- | The bitwise exclusive or of two values.

   > toInt (bitXor (-1) 5)
   Some -6 -}
export
bitXor : I64 -> I64 -> I64
bitXor a b = i64FromBits (u64BitXor (i64ToBits a) (i64ToBits b))

{- | Every bit of the value inverted, so `bitNot x` is `-x - 1`.

   > toInt (bitNot 0)
   Some -1 -}
export
bitNot : I64 -> I64
bitNot x = i64FromBits (u64BitXor (i64ToBits x) 0xFFFFFFFFFFFFFFFF)

{- | The value shifted left by a number of bits, the vacated bits zero.

   Bits shifted past bit 63 are discarded, the sign bit included, and a shift
   of 64 or more gives `0`. A negative shift panics.

   > shiftLeft 1 63 == minBound
   True
   > toInt (shiftLeft 1 64)
   Some 0 -}
export
shiftLeft : I64 -> Int -> I64
shiftLeft x k
  | k < 0 = panic "i64.shiftLeft: negative shift \{k}"
  | otherwise = i64FromBits (u64ShiftLeft (i64ToBits x) k)

{- | The value shifted right by a number of bits, the vacated bits copies of
   the sign bit, so a negative value stays negative.

   A shift of 64 or more gives `0`, or `-1` for a negative value. A negative
   shift panics.

   > toInt (shiftRight (-16) 2)
   Some -4
   > toInt (shiftRight minBound 70)
   Some -1 -}
export
shiftRight : I64 -> Int -> I64
shiftRight x k
  | k < 0 = panic "i64.shiftRight: negative shift \{k}"
  | x < 0 = bitNot (i64FromBits (u64ShiftRight (i64ToBits (bitNot x)) k))
  | otherwise = i64FromBits (u64ShiftRight (i64ToBits x) k)

-- > toInt (shiftRight 16 70)
-- Some 0
-- > toInt (shiftRight minBound 63)
-- Some -1

-- # Instances

export impl Eq I64 where
  eq a b = a == b

-- Every method is defined, not only `compare`, so each one is a single builtin
-- comparison and a generic caller never falls back to an interface default.
-- `min` and `max` are the methods being defined, so they are written out.
-- lint-disable-next-line rule-if-max-min
export impl Ord I64 where
  compare a b = if a < b then Lt else if a > b then Gt else Eq
  lt a b = a < b
  gt a b = a > b
  lte a b = a <= b
  gte a b = a >= b
  min a b = if a <= b then a else b
  max a b = if a >= b then a else b

{- | Arithmetic modulo 2^64 in two's complement. `fromInt` never panics, since
   every `Int` fits. `abs` and `negate` of `minBound` are `minBound`. -}
export impl Num I64 where
  add a b = a + b
  sub a b = a - b
  mul a b = a * b
  div a b = a / b
  negate a = 0 - a
  abs a = if a < 0 then 0 - a else a
  signum a = if a > 0 then 1 else if a < 0 then 0 - 1 else 0
  fromInt n = ofInt n
  rem a b = a % b

export impl Bounded I64 where
  minBound = -9223372036854775808
  maxBound = 9223372036854775807

-- The hash of the 64 bits as a `U64`, so an `I64` from `0` to `2^62 - 1`
-- hashes as that `Int` does.
export impl Hashable I64 where
  hash x = hash (i64ToBits x)

-- A negative value is the sign and its magnitude, and `minBound`'s magnitude
-- 2^63 is the `U64` its negation's bits spell.
export impl Display I64 where
  display x =
    if x < 0 then "-" ++ display (i64ToBits (0 - x)) else display (i64ToBits x)

export impl Debug I64 where
  debug x = display x
